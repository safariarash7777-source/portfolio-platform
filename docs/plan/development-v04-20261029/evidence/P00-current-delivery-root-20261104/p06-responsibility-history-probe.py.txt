def run(args,body=None):

 if r.returncode or len(r.stdout.encode())>65536:raise RuntimeError('BOUNDED_READ_UNCONFIRMED')
 return r.stdout

 assert r['id']==SOURCE and r['image']==IMAGE and r['health']=='healthy'
